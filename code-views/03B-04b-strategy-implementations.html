Strategy pattern · Three channels

Email (Laravel Mail), Push (Firebase FCM), and SMS (Twilio) — three interchangeable strategies. Adding a fourth is one new class + one match arm.

app/Services/Notification/{Email,Push,Sms}Channel.php PHP 
// ── app\Services\Notification\EmailChannel.php ──
<?php

namespace App\Services\Notification;

use App\Models\User;
use App\Services\Notification\Contracts\NotificationChannel;
use Illuminate\Support\Facades\Mail;

class EmailChannel implements NotificationChannel
{
    public function send(int $userId, string $message): void
    {
        $user = User::findOrFail($userId);

        Mail::raw($message, function ($mail) use ($user) {
            $mail->to($user->email)->subject('Lifestyle notification');
        });
    }

    public function name(): string
    {
        return 'email';
    }
}


// ── app\Services\Notification\PushChannel.php ──
<?php

namespace App\Services\Notification;

use App\Services\Notification\Contracts\NotificationChannel;
use Illuminate\Support\Facades\Http;

class PushChannel implements NotificationChannel
{
    /** Firebase Cloud Messaging (FCM) endpoint. */
    private const FCM_ENDPOINT = 'https://fcm.googleapis.com/fcm/send';

    public function send(int $userId, string $message): void
    {
        Http::withToken(config('services.fcm.server_key'))
            ->post(self::FCM_ENDPOINT, [
                'to'           => "/topics/user_{$userId}",
                'notification' => [
                    'title' => 'Lifestyle',
                    'body'  => $message,
                ],
            ]);
    }

    public function name(): string
    {
        return 'push';
    }
}


// ── app\Services\Notification\SmsChannel.php ──
<?php

namespace App\Services\Notification;

use App\Models\User;
use App\Services\Notification\Contracts\NotificationChannel;
use Illuminate\Support\Facades\Http;

class SmsChannel implements NotificationChannel
{
    public function send(int $userId, string $message): void
    {
        $user = User::findOrFail($userId);

        // Twilio-compatible payload
        Http::withBasicAuth(
                config('services.twilio.sid'),
                config('services.twilio.token'),
            )
            ->asForm()
            ->post('https://api.twilio.com/2010-04-01/Accounts/'
                . config('services.twilio.sid') . '/Messages.json', [
                'From' => config('services.twilio.from'),
                'To'   => $user->phone ?? '',
                'Body' => $message,
            ]);
    }

    public function name(): string
    {
        return 'sms';
    }
}
