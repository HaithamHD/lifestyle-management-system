§ 3B.1 · Repository pattern (1/3)

Repository pattern · Contract

Controllers depend on this interface, never on Eloquent directly. This is the abstraction seam that makes tests trivial to mock and lets us swap the persistence technology in one line.

app/Repositories/Contracts/TaskRepositoryInterface.php PHP 
<?php

namespace App\Repositories\Contracts;

use App\Models\Task;
use Illuminate\Support\Collection;

/**
 * Repository Pattern — Contract.
 *
 * Controllers depend on this interface, never on Eloquent directly.
 * Benefits:
 *   • Persistence layer can be swapped (SQL → NoSQL) without touching controllers.
 *   • Unit tests can bind a fake implementation in the container.
 *   • Query complexity is centralized in one class per aggregate root.
 */
interface TaskRepositoryInterface
{
    /** Return tasks for a user filtered by preset (today/upcoming/completed/overdue/all). */
    public function forUser(int $userId, string $filter = 'today'): Collection;

    public function create(int $userId, array $data): Task;

    public function update(Task $task, array $data): Task;

    public function delete(Task $task): void;

    /** Percentage of tasks completed this ISO week (0..100). */
    public function weeklyCompletionScore(int $userId): int;

    public function weeklyCompletedCount(int $userId): int;

    public function weeklyTotalCount(int $userId): int;

    /** [ ['week' => 'Wxx', 'score' => int], … ]  for the last N weeks. */
    public function weeklyHistory(int $userId, int $weeks = 4): array;
}
